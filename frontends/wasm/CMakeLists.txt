cmake_minimum_required(VERSION 3.25)
project(blytplay C)

# -------------------------------------------------------------------------
# Guard: this project must be configured with Emscripten.  Normally configured
# by the SDK build (cmake --build build --target sdk) at build/build-wasm and
# build/build-wasm-debug; rebuild incrementally with cmake --build <that dir>.
# -------------------------------------------------------------------------
if(NOT EMSCRIPTEN)
  message(
    FATAL_ERROR
      "blytplay (WASM) must be built with Emscripten.\n"
      "Build via the SDK target (configures this tree automatically):\n"
      "  cmake --build build --target sdk\n"
      "or configure manually with:\n"
      "  emcmake cmake -B <dir> -S frontends/wasm")
endif()

set(CMAKE_C_STANDARD 11)

# Repository root (two levels above frontends/wasm/)

# Version: prefer the stamped build/version.txt from the host build

else()
  file(READ "${BLYT_ROOT}/version.txt" BLYT_VERSION)
endif()

# -------------------------------------------------------------------------
# Pre-generated FlatBuffers C reader headers (from the host cmake build)
#
# The WASM build does not run flatcc_cli (a host-native tool). The reader
# headers generated by the host build are platform-independent and can be reused
# here directly.
# -------------------------------------------------------------------------
set(GENERATED_DIR
    "${BLYT_ROOT}/build/generated/flatbuffers"
    CACHE PATH
          "Pre-generated FlatBuffers C reader headers (from host cmake build)")
if(NOT EXISTS "${GENERATED_DIR}/cart_info_reader.h")
  message(

      "FlatBuffers reader headers not found at ${GENERATED_DIR}.\n"
      "Run the host cmake build first:\n"
      "  cmake -B build && cmake --build build")
endif()
message(STATUS "blyt WASM: FlatBuffers headers from ${GENERATED_DIR}")

# -------------------------------------------------------------------------
# Pre-built RV32IMAFC guest libraries
#
# libblytcommon.so, libblytc.so, and libblyt32.so are RV32IMAFC binaries built
# by the host cmake build (clang --target=riscv32).  They are embedded into the
# WASM module as C byte arrays via bin2c.py so dynlink can load them from memory
# without a filesystem path.
# -------------------------------------------------------------------------
set(BLYT_GUEST_LIB_DIR
    "${BLYT_ROOT}/build/sdk/lib"
    CACHE PATH
          "Directory containing libblytcommon.so, libblytc.so, libblyt32.so")

foreach(_LIB libblytcommon.so libblytc.so libblyt32.so)
  if(NOT EXISTS "${BLYT_GUEST_LIB_DIR}/${_LIB}")
    message(
      FATAL_ERROR
        "Guest library not found: ${BLYT_GUEST_LIB_DIR}/${_LIB}\n"
        "Build the host SDK first:\n"
        "  cmake -B build && cmake --build build --target sdk")
  endif()
endforeach()
message(STATUS "blyt WASM: guest libs from ${BLYT_GUEST_LIB_DIR}")

# -------------------------------------------------------------------------
# Optional Lua-direct support
#
# When Lua sources are available, compile the Lua VM sources and the blyt32lua
# bindings directly into the WASM module so Lua carts run without going through
# rv32emu.  BLYT_LUA_SOURCE_DIR is set by blyt_sdk.cmake (which gets it from
# FetchContent); falls back to third_party/lua for local dev.
# -------------------------------------------------------------------------
if(NOT DEFINED BLYT_LUA_SOURCE_DIR)
  set(BLYT_LUA_SOURCE_DIR "${BLYT_ROOT}/third_party/lua")
endif()
set(LUA_DIR "${BLYT_LUA_SOURCE_DIR}")
set(BLYT_LUA_SOURCES "")
if(EXISTS "${LUA_DIR}/lvm.c")
  file(GLOB LUA_ALL_SRCS "${LUA_DIR}/*.c")
  # Exclude the standalone interpreter, compiler, and sandboxed-out libs. utf8
  # is KEPT (ADR-0079, issue #167) so the host-Lua fast path matches the guest
  # build.
  foreach(
    _EXCL
    "${LUA_DIR}/lua.c"
    "${LUA_DIR}/luac.c"
    "${LUA_DIR}/onelua.c"
    "${LUA_DIR}/liolib.c"
    "${LUA_DIR}/loslib.c"
    "${LUA_DIR}/loadlib.c"
    "${LUA_DIR}/ldblib.c")
    list(REMOVE_ITEM LUA_ALL_SRCS "${_EXCL}")
  endforeach()
  set(BLYT_LUA_SOURCES ${LUA_ALL_SRCS})
  message(STATUS "blyt WASM: Lua-direct support enabled (${LUA_DIR})")
else()
  message(
    STATUS
      "blyt WASM: Lua-direct support disabled (Lua sources not found at ${LUA_DIR})"
  )
endif()

# -------------------------------------------------------------------------
# DAP debugging support (Lua source-level via Debug Adapter Protocol)
#
# When ON: compile dap_transport_wasm.c + master_hook.c into blytplay and define
# BLYT_DAP=1.  Pausing the Lua VM uses Lua coroutine yield (not ASYNCIFY), so no
# special link options are needed.
# -------------------------------------------------------------------------
set(BLYT_DAP_DIR "${BLYT_ROOT}/runtime/host/src/dap")
option(BLYT_DAP "Enable DAP Lua debugger support" ON)

set(BLYT_DAP_SOURCES "")
set(BLYT_DAP_DEFS "")
set(BLYT_DAP_LINK_OPTS "")
if(BLYT_DAP AND EXISTS "${LUA_DIR}/lvm.c")
  list(APPEND BLYT_DAP_SOURCES "${BLYT_DAP_DIR}/dap_transport_wasm.c"
       "${BLYT_DAP_DIR}/master_hook.c")
  list(APPEND BLYT_DAP_DEFS BLYT_DAP=1)
  # wdap_dequeue_json (EM_JS) calls _malloc/_free from JavaScript.
  list(APPEND BLYT_DAP_LINK_OPTS "-sEXPORTED_FUNCTIONS=_main,_malloc,_free")
  message(STATUS "blyt WASM: DAP debugging enabled")
else()
  if(BLYT_DAP AND NOT EXISTS "${LUA_DIR}/lvm.c")
    message(
      STATUS
        "blyt WASM: DAP disabled (requires BLYT_LUA; Lua sources not found at ${LUA_DIR})"
    )
  else()
    message(STATUS "blyt WASM: DAP disabled")
  endif()
endif()

# -------------------------------------------------------------------------
# GDB RSP debugging support (WebSocket transport)
#
# When ON: compile gdb_stub.c + gdb_transport_wasm.c into blytplay and define
# BLYT_GDB=1.  The WebSocket transport is non-blocking; the emulator polls each
# animation tick via blyt_session_run_frame (no threads needed).
# -------------------------------------------------------------------------
set(BLYT_GDB_DIR "${BLYT_ROOT}/runtime/host/src/gdb")
option(BLYT_GDB "Enable GDB RSP debugger support (WebSocket transport)" ON)

set(BLYT_GDB_SOURCES "")
set(BLYT_GDB_DEFS "")
set(BLYT_GDB_LINK_OPTS "")
if(BLYT_GDB AND EXISTS "${BLYT_GDB_DIR}/gdb_transport_wasm.c")
  list(APPEND BLYT_GDB_SOURCES "${BLYT_GDB_DIR}/gdb_stub.c"
       "${BLYT_GDB_DIR}/gdb_transport_wasm.c")
  list(APPEND BLYT_GDB_DEFS BLYT_GDB=1)
  # wgdb_dequeue_pkt (EM_JS) calls _malloc/_free from JavaScript.
  list(APPEND BLYT_GDB_LINK_OPTS "-sEXPORTED_FUNCTIONS=_main,_malloc,_free")
  message(STATUS "blyt WASM: GDB RSP debugging enabled")
else()
  if(BLYT_GDB AND NOT EXISTS "${BLYT_GDB_DIR}/gdb_transport_wasm.c")
    message(STATUS "blyt WASM: GDB disabled (gdb_transport_wasm.c not found)")
  else()
    message(STATUS "blyt WASM: GDB disabled")
  endif()
endif()

# -------------------------------------------------------------------------
# Embed guest libraries as C byte arrays (same approach as blyt_libretro)
# -------------------------------------------------------------------------
set(EMBEDDED_LIBS_C "${CMAKE_BINARY_DIR}/blytplay_embedded_libs.c")
set(LIBBLYTCOMMON "${BLYT_GUEST_LIB_DIR}/libblytcommon.so")
set(LIBBLYTC "${BLYT_GUEST_LIB_DIR}/libblytc.so")
set(LIBBLYT32 "${BLYT_GUEST_LIB_DIR}/libblyt32.so")
# ADR-0130: embed the ECALL-bridge variant of libblyt32lua.so when present
# (registered under the same DT_NEEDED name).  The real Lua-VM guest library is
# dead weight on this target — host-side Lua is authoritative — and the bridge
# variant gives guest wrappers a working Lua C API via ECALL 10.
if(EXISTS "${BLYT_GUEST_LIB_DIR}/libblyt32lua-bridge.so")
  set(LIBBLYT32LUA "${BLYT_GUEST_LIB_DIR}/libblyt32lua-bridge.so")
else()
  set(LIBBLYT32LUA "${BLYT_GUEST_LIB_DIR}/libblyt32lua.so")
endif()

add_custom_command(
  OUTPUT "${EMBEDDED_LIBS_C}"
  COMMAND
    python3 "${BLYT_ROOT}/cmake/bin2c.py" blytcommon_so "${LIBBLYTCOMMON}"
    blytc_so "${LIBBLYTC}" blyt32_so "${LIBBLYT32}" blyt32lua_so
    "${LIBBLYT32LUA}" "${EMBEDDED_LIBS_C}"
  DEPENDS "${LIBBLYTCOMMON}" "${LIBBLYTC}" "${LIBBLYT32}" "${LIBBLYT32LUA}"
  COMMENT "Embedding guest libs for blytplay (WASM)"
  VERBATIM)

# Softfloat compile options: INLINE has a space so it must be a compile option
# (not a cmake definition) to avoid quoting issues. Force-include rv32emu
# common.h for rv32emu and softfloat sources only. Applying it globally
# conflicts with Lua's UNUSED macro: rv32emu/common.h redefines UNUSED as
# __attribute__((unused)) (no args), which breaks Lua's UNUSED(x) call-style
# macro.
foreach(_SRC ${RV32EMU_SOURCES} ${SOFTFLOAT_SOURCES})
  set_source_files_properties(
    "${_SRC}" PROPERTIES COMPILE_OPTIONS "-include;${RV32EMU_CONFIG_H}")
endforeach()

# Emscripten link options
target_link_options(
  blytplay
  PRIVATE
  ${BLYT_DAP_LINK_OPTS}
  ${BLYT_GDB_LINK_OPTS}
  # Allow WASM linear memory to grow to accommodate the 256 MiB guest region.
  "-sALLOW_MEMORY_GROWTH=1"
  # 64 MiB initial allocation; grows when blyt_session_create mallocs the 256
  # MiB guest address space.
  "-sINITIAL_MEMORY=67108864"
  # 256 KiB stack — rv32emu instruction dispatch uses significant call depth.
  # The large blyt_symtab_t (~66 KiB) is heap-allocated so this headroom is
  # sufficient, but keep well above the 64 KiB default.
  "-sSTACK_SIZE=262144"
  # Support both browser and Node.js environments so the same binary can be
  # tested headlessly under Node (integration test wasm_cart_runs_in_node).
  "-sENVIRONMENT=web,node"
  # Ensure the Emscripten FS API is available (cart is loaded via FS.writeFile
  # in shell.html before main() runs).
  "-sFORCE_FILESYSTEM=1"
  # Runtime methods used by the page glue: addRunDependency/removeRunDependency
  # gate the cart preload; ccall lets the dev control channel (issue #87) pass
  # JSON command strings into the C handler; FS lets the reload flow rewrite
  # /cart.blyt in MEMFS.
  "-sEXPORTED_RUNTIME_METHODS=ccall,FS,addRunDependency,removeRunDependency"
  # Allow the runtime to exit cleanly when the cart finishes.
  "-sNO_EXIT_RUNTIME=0"
  # Custom HTML shell with canvas + cart preload logic.
  "--shell-file=${CMAKE_CURRENT_SOURCE_DIR}/shell.html"
  # Pre-js: initialise Module from globalThis.__blyt_module_init when set by the
  # Node.js test driver (tests/wasm/run_cart.js).  Placed inside the IIFE before
  # Emscripten's own var Module assignment so the driver's preRun / onExit hooks
  # are picked up.
  "--pre-js=${CMAKE_CURRENT_SOURCE_DIR}/module_pre.js")
